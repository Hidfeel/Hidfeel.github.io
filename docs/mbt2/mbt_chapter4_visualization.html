FUNCTION run_Assessment(patient): // 优先评估领域 assess(patient.self_function) assess(patient.mentalizing_vulnerabilities) assess(patient.trigger_contexts) assess(patient.risk_factors) // 特定人群适配 IF patient.has_narcissistic_traits: PRIORITIZE assess(patient.self_other_representation) ELSE IF patient.has_antisocial_PD: ADD assess(patient.violence_link_to_mentalizing_loss) ELSE IF patient.has_depression_with_somatoform: ADD assess(patient.somatic_markers) END IF // 线上 MBT 额外步骤 IF modality == "online": IF possible: schedule_face_to_face_meeting_first() early_appraise(patient.attachment_anxiety) adjust_interaction_style_accordingly() END IF RETURN assessment_results END FUNCTION

FUNCTION co_create_Formulation(patient, clinician): // Step 1-4: 起草→解释→改写→协议 draft = clinician.draft_formulation(language="patient_meaningful") clinician.explain(draft); collaboratively_discuss(draft) patient.review_and_rewrite(draft, support_if_needed) agreement = JOINTLY_AGREE_ON(focus="session_focus") // 持续更新 WHILE treatment_in_progress: IF new_material_emerges: UPDATE formulation END IF END WHILE // 入组准备 IF entering_group: relational_passport = summarize(attendance_rationale, emphasis=relational_difficulties) IF assessor != group_clinician: hold_tripartite_meeting() END IF END IF RETURN collaborative_agreement, relational_passport END FUNCTION

FUNCTION session_Intervention(patient, clinician, formulation): // Phase 1: 共情探索 troubling_issue = identify_current_trouble(patient) explore(troubling_issue, context = formulation) // Phase 2: 共情确认（首选干预） SHOW_RECOGNITION_AND_RESONANCE(patient.emotion_then_and_now) EMPATHIZE_WITH(primary_emotions) IDENTIFY_AND_RECOGNIZE(effect_on_patient) // Phase 3: 不知立场探索 expand_perspectives(patient.narrative) theme = gauge_focus_for_further_work() IF theme NOT linked_to_formulation: ASK patient TO link_subject_to_formulation() IF still_unclear: REVISE formulation END IF END IF // Phase 4-8 ... affect_identification(); contextualization() explore_alternative_perspectives() IF alliance_rupture: repair_rupture() END IF IF low_mentalizing_stabilized: focus_on_content() END IF sit_side_by_side(); collaborate_on_alternative_perspectives() MONITOR patient.reaction; MONITOR clinician.reaction END FUNCTION

核心临床者原则

贯穿所有流程

✓
识别无效的低水平心智化
✗
不为患者代劳心智化
✗
不对非心智化话语进行心智化
✗
不参与空洞的表面心智化
✓
监测当前心智化（过程 vs 内容）
✗
不附和低心智化
✓
管理焦虑，滴定依恋焦虑
✓
平衡心智化：稳定→跟随情感；不稳定→相反移动
✓
激活关系过程：去激活↔超激活
✓
顺应患者和自身反应性
✓
保持定期督导/同辈讨论
